cmake_minimum_required(VERSION 3.16)
project(NeuralPLSI LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(neuralplsi SHARED core.cpp spline.cpp)
target_link_libraries(neuralplsi PRIVATE Threads::Threads)
target_compile_features(neuralplsi PUBLIC cxx_std_17)
target_include_directories(neuralplsi PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(neuralplsi PRIVATE -O3 -ffp-contract=off -Wall -Wextra)
endif()
add_executable(neuralplsi_demo ../../examples/native_neuralplsi.cpp)
target_link_libraries(neuralplsi_demo PRIVATE neuralplsi)
